    [Serializable] sealed class RunResult { public bool pass; public string message, utc; }
    const string Flag = "OutOfReach.BarePalmPlaneQa", OutputKey = Flag + ".Output", Deadline = Flag + ".Deadline", Errors = Flag + ".Errors";
    public const string ScenePath = "Assets/_OutOfReach/Scenes/Bootstrap.unity"; // source provenance only; fixture is unsaved
    public static string Output => SessionState.GetString(OutputKey, Path.GetFullPath(".local/bare-palm-plane-qa"));
    public static bool HitchMode => SessionState.GetBool(Flag + ".Hitch", false);
    public static bool HitchHeldOnly => SessionState.GetBool(Flag + ".HitchHeldOnly", false);
    static BarePalmPlaneQa()
    {
        EditorApplication.playModeStateChanged += OnPlayMode;
        EditorApplication.update += Watchdog;
        Application.logMessageReceived += OnLog;
    }
    public static void Run() => RunCore(false);
    public static void RunHitch() => RunCore(true);
    public static void RunHitchHeld() => RunCore(true,true);
    static void RunCore(bool hitch,bool heldOnly=false)
    {
        try
        {
            string project = Path.GetFullPath(Path.Combine(Application.dataPath, "..")).Replace('\\', '/');
            if (!Application.isBatchMode) throw new InvalidOperationException("Batch-only runner.");
            if (project.IndexOf("/.local/", StringComparison.OrdinalIgnoreCase) < 0) throw new InvalidOperationException("Run only in the isolated .local project.");
            if (Application.unityVersion != "6000.3.28f1") throw new InvalidOperationException("Unity 6000.3.28f1 required.");
            if (SystemInfo.graphicsDeviceType == GraphicsDeviceType.Null) throw new InvalidOperationException("Real GPU captures require graphics; omit -nographics.");
            if (EditorApplication.isPlayingOrWillChangePlaymode) throw new InvalidOperationException("Start in Edit mode.");
            SessionState.SetBool(Flag + ".Hitch", hitch);
            SessionState.SetBool(Flag + ".HitchHeldOnly", heldOnly);
            SessionState.SetString(OutputKey, Path.Combine(project, hitch ? ".local/contact-hitch-qa" : ".local/bare-palm-plane-qa", DateTime.UtcNow.ToString("yyyyMMddTHHmmssfffZ")));
            Directory.CreateDirectory(Output);
            SessionState.SetString(Errors, "");
            SessionState.SetString(Deadline, DateTime.UtcNow.AddSeconds(100).ToString("O", CultureInfo.InvariantCulture));
            SessionState.SetString(Flag, "entering");
            File.WriteAllText(Path.Combine(Output, "scope.txt"),
                "Unsaved bounded wall/rail fixture, NOT Bootstrap/full-game/input-binding acceptance.\n"
                + "One real Player and MissionSession. Real launch/cockpit/repairs/suit command; actual pending Impact acknowledged skipped.\n"
                + "Real EnterFloat at a visible nearby solid. Public Teleport/ClearMotion reset positions BETWEEN cases only. Scripted zero device intent.\n"
                + "No synthetic phase/suit/mode/inventory; session is not auto-ticked during geometry probes. Body authoring reuses imported released models.\n"
                + "Closest-hit validator plus 10 mm padded wrist path; NOT whole-arm collision/IK acceptance. All material-defined bare skin rebaked on every rendered active/return frame and at callback.\n"
                + "1280x720 actual camera, endCameraRendering capture on contact frame. Native art acceptance still needs human viewing. 100s watchdog.\n");
            if (hitch) File.AppendAllText(Path.Combine(Output, "scope.txt"),
                "CONTACT-HITCH: real 160ms main-thread stalls after presenter LateUpdate, then observe actual next-frame delta/pose/stage/action counts.\n"
                + "No private presentation advance is invoked. Private elapsed is read-only instrumentation. All original geometry validation remains unchanged.\n"
                + "Initial contact GPU readback is suppressed in hitch mode to isolate the injected stall; measured contact JSON remains. Two PNGs show held grips AFTER the stall.\n"
                + "WORK/glove approaches retain animation time and issue no hitch-frame action; static grips remain, geometry jumps/occlusion still cancel.\n");
            EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
            BuildFixture();
            WriteManifest();
            EditorApplication.isPlaying = true;
        }
        catch (Exception error) { Finish(false, error.ToString()); }
    }
    static void OnPlayMode(PlayModeStateChange state)
    {
        if (state != PlayModeStateChange.EnteredPlayMode || SessionState.GetString(Flag, "") != "entering") return;
        SessionState.SetString(Flag, "running");
        new GameObject("Contact native acceptance - QA only").AddComponent<BarePalmPlaneQaDriver>();
    }
    static void Watchdog()
    {
        if (string.IsNullOrEmpty(SessionState.GetString(Flag, ""))) return;
        if (DateTime.TryParse(SessionState.GetString(Deadline, ""), CultureInfo.InvariantCulture, DateTimeStyles.RoundtripKind, out DateTime deadline) && DateTime.UtcNow > deadline)
            Finish(false, "100-second watchdog expired. Last completed assertion is in checks.json/checks.txt.");
    }
    static void OnLog(string message, string stack, LogType type)
    {
        if (string.IsNullOrEmpty(SessionState.GetString(Flag, "")) || (type != LogType.Error && type != LogType.Exception && type != LogType.Assert)) return;
        string entry = type + ": " + message + "\n" + stack + "\n";
        // Preserve the same narrowly identified Editor Search startup issue separately; never ignore game errors.
        if (type == LogType.Exception && message.StartsWith("ArgumentOutOfRangeException:", StringComparison.Ordinal)
            && stack.Contains("UnityEditor.Search.SearchDatabase") && stack.Contains("UnityEditor.Search.SearchInit.IndexationOnStartup") && !stack.Contains("OutOfReach"))
        { File.AppendAllText(Path.Combine(Output, "editor-environment-errors.txt"), entry); return; }
        SessionState.SetString(Errors, SessionState.GetString(Errors, "") + entry);
        File.AppendAllText(Path.Combine(Output, "runtime-errors.txt"), entry);
    }
    public static void RequireNoErrors()
    {
        string errors = SessionState.GetString(Errors, "");
        if (!string.IsNullOrEmpty(errors)) throw new InvalidOperationException("Runtime errors:\n" + errors);
    }
    public static void Finish(bool pass, string message)
    {
        SessionState.SetString(Flag, "");
        Directory.CreateDirectory(Output);
        File.WriteAllText(Path.Combine(Output, pass ? "PASS.txt" : "FAIL.txt"), message);
        File.WriteAllText(Path.Combine(Output, "result.json"), JsonUtility.ToJson(new RunResult { pass = pass, message = message, utc = DateTime.UtcNow.ToString("O") }, true));
        Debug.Log((pass ? "CONTACT_NATIVE_QA_PASS " : "CONTACT_NATIVE_QA_FAIL ") + message + " output=" + Output);
        EditorApplication.Exit(pass ? 0 : 1);
    }
    static void WriteManifest()
    {
        var paths = Directory.GetFiles("Assets/_OutOfReach", "*", SearchOption.AllDirectories)
            .Where(p => p.EndsWith(".cs") || p.EndsWith(".fbx") || p.EndsWith(".prefab") || p.EndsWith(".json") || p.EndsWith(".mat") || p.EndsWith(".asmdef"))
            .Concat(Directory.GetFiles("Assets/Editor", "BarePalmPlaneQa.cs", SearchOption.AllDirectories))
            .Concat(new[] { ScenePath, "ProjectSettings/ProjectVersion.txt" }).OrderBy(p => p, StringComparer.Ordinal);
        using (var sha = SHA256.Create())
            File.WriteAllLines(Path.Combine(Output, "source-sha256.txt"), paths.Select(p =>
                BitConverter.ToString(sha.ComputeHash(File.ReadAllBytes(p))).Replace("-", "").ToLowerInvariant() + "  " + p.Replace('\\', '/')));
    }
    static void BuildFixture()
    {
        var rig=new GameObject("Contact QA player");
        var cc=rig.AddComponent<CharacterController>(); cc.height=1.8f; cc.radius=.3f; cc.center=new Vector3(0,.9f,0); cc.skinWidth=.03f; cc.stepOffset=.3f;
        Transform eye=Child("Unbobbed reach origin",rig.transform); eye.localPosition=new Vector3(0,1.65f,0);
        var resolver=eye.gameObject.AddComponent<InteractionResolver>(); resolver.Configure(eye,rig.transform,.85f);
        var player=rig.AddComponent<FirstPersonController>(); player.Configure(eye,resolver);
        Camera camera=Child("Contact QA camera",eye).gameObject.AddComponent<Camera>();
        camera.tag="MainCamera"; camera.fieldOfView=68; camera.nearClipPlane=.05f; camera.farClipPlane=30;
        camera.clearFlags=CameraClearFlags.SolidColor; camera.backgroundColor=new Color(.025f,.035f,.045f); camera.allowHDR=false; camera.allowMSAA=false;
        camera.gameObject.AddComponent<AudioListener>();
        Transform presentation=Child("Contact QA presentation",null);
        I06BodyAuthoring.Create(presentation,rig.transform,camera);
        const string path="Assets/_OutOfReach/Visuals/Character/FirstPersonBody/FloatNeutralV04/Animations/BODY_FloatNeutral_V04.fbx";
        using(var sha=SHA256.Create())
            if(BitConverter.ToString(sha.ComputeHash(File.ReadAllBytes(path))).Replace("-","").ToLowerInvariant()
                !="63b33733541644cfbbd1e16d94d3ece44601de95b2829406039f111efffb37d8")
                throw new InvalidOperationException("Import the final rest-bind V04 clip with its approved hash first.");
        var importer=AssetImporter.GetAtPath(path) as ModelImporter;
        if(importer==null||importer.animationType!=ModelImporterAnimationType.Generic||importer.optimizeGameObjects||!importer.preserveHierarchy
            ||importer.animationCompression!=ModelImporterAnimationCompression.Off)
            throw new InvalidOperationException("V04 requires Generic/unoptimized/uncompressed/preserveHierarchy import; root owns that asset change.");
        var clip=AssetDatabase.LoadAllAssetsAtPath(path).OfType<AnimationClip>().SingleOrDefault(c=>!c.name.StartsWith("__preview__",StringComparison.Ordinal));
        if(!clip||clip.legacy)throw new InvalidOperationException("Exactly one non-legacy V04 clip is required.");
        var body=Object.FindFirstObjectByType<EmbodiedBodyPresenter>();
        var serialized=new SerializedObject(body);
        serialized.FindProperty("work").FindPropertyRelative("floating").objectReferenceValue=clip;
        serialized.FindProperty("eva").FindPropertyRelative("floating").objectReferenceValue=clip;
        serialized.ApplyModifiedPropertiesWithoutUndo();
        body.SetVisualFeetOffset(EmbodiedBodyPresenter.BodyPose.Floating,Vector3.zero); // unsaved fixture instance only

        var prefab=AssetDatabase.LoadAssetAtPath<GameObject>(I01BootstrapBuilder.HandsPath);
        var json=AssetDatabase.LoadAssetAtPath<TextAsset>(I01BootstrapBuilder.HandsRigPath);
        if (!prefab||!json) throw new InvalidOperationException("Original glove prefab/rig missing.");
        Transform mount=Child("Original glove mount",camera.transform);
        var model=(GameObject)PrefabUtility.InstantiatePrefab(prefab,mount);
        model.transform.SetLocalPositionAndRotation(Vector3.zero,Quaternion.identity); model.transform.localScale=Vector3.one;
        presentation.gameObject.AddComponent<AstronautHandsPresenter>().ConfigureModel(model.transform,mount,json);
        presentation.gameObject.AddComponent<BareArmContactPresenter>();
        RenderSettings.ambientMode=AmbientMode.Flat; RenderSettings.ambientLight=new Color(.42f,.46f,.50f);
        var key=Child("Fixture key",null).gameObject.AddComponent<Light>(); key.type=LightType.Directional; key.intensity=1.6f; key.transform.rotation=Quaternion.Euler(35,-25,0);
        var fill=Child("Fixture fill",null).gameObject.AddComponent<Light>(); fill.type=LightType.Point; fill.intensity=3; fill.range=5; fill.transform.position=new Vector3(-.6f,1.9f,-.5f);
    }
    static Transform Child(string name,Transform parent) { var t=new GameObject(name).transform; if(parent)t.SetParent(parent,false); return t; }

}

    bool Control()=>session!=null&&session.IsPlayerControlAllowed;

    bool BareLease(ContactHand hand)=>Lease()&&!session.Eva.IsSuitOn&&body.IsVisible&&body.CurrentPose==EmbodiedBodyPresenter.BodyPose.Floating;
    IEnumerator Start()
    {
        originalBackground=Application.runInBackground;Application.runInBackground=true;
        var stack=new Stack<IEnumerator>();stack.Push(Exercise());
        while(stack.Count>0)
        {
            bool more=false;object current=null;Exception failure=null;
            try {BarePalmPlaneQa.RequireNoErrors();if(captureError!=null)throw new InvalidOperationException(captureError);more=stack.Peek().MoveNext();if(more)current=stack.Peek().Current;}
            catch(Exception e){failure=e;}
            if(failure!=null)
            {
                foreach(var it in stack)try{(it as IDisposable)?.Dispose();}catch{}
                Add(false,failure.ToString());Cleanup();BarePalmPlaneQa.Finish(false,"Stage: "+stage+"\n"+failure);yield break;
            }
            if(!more){stack.Pop();continue;}if(current is IEnumerator child){stack.Push(child);continue;}yield return current;
        }
        Cleanup();BarePalmPlaneQa.Finish(true,checks.checks.Count+" checks passed; "+captures
            +(BarePalmPlaneQa.HitchMode?" post-stall held-grip captures.":" contact-frame and "+lookDownCaptures+" look-down captures.")+" Bounded fixture only; see scope.txt.");
    }
    IEnumerator Exercise()
    {
        player=Object.FindFirstObjectByType<FirstPersonController>();body=Object.FindFirstObjectByType<EmbodiedBodyPresenter>();
        gloves=Object.FindFirstObjectByType<AstronautHandsPresenter>();bare=Object.FindFirstObjectByType<BareArmContactPresenter>();cameraView=Camera.main;eye=player.ViewPivot;
        Check(Object.FindObjectsByType<FirstPersonController>(FindObjectsSortMode.None).Length==1&&Object.FindObjectsByType<CharacterController>(FindObjectsSortMode.None).Length==1,"Exactly one real Player/capsule.");
        Check(Object.FindObjectsByType<Camera>(FindObjectsSortMode.None).Length==1&&body&&gloves&&bare,"One fixture camera and released body/glove/bare presenters.");
        quietInput=new QuietInput();player.InputSource=quietInput;player.SendMessage("OnApplicationFocus",true,SendMessageOptions.RequireReceiver);
        session=new MissionSession(subscriberErrorSink: e=>Debug.LogException(e));player.ControlGate=Control;
        Cmd(session.Planner.TryLaunch());Cmd(session.Director.TryCompleteCockpitCheck());
        Check(session.Cinematics.IsWaiting&&session.Cinematics.ActiveCinematic==CinematicId.Impact,"Real Impact requested by cockpit command.");
        session.Events.Publish(new CinematicFinished(CinematicId.Impact,session.Cinematics.ActiveSequenceId,true));
        Cmd(session.Systems.Power.TryRepair(session.Resources));Cmd(session.Systems.LifeSupport.TryRepair(session.Resources));
        Check(session.State.Phase==GamePhase.Repair&&!session.Eva.IsSuitOn&&session.IsPlayerControlAllowed,"Actual repaired, unsuited session enables fixture control.");
        body.ConfigureSession(session.Eva,session.ClockView,()=>session!=null&&!session.State.IsClosed,()=>player.IsFloating,
            ()=>session.Eva.IsSuitOn?EmbodiedBodyPresenter.BodyArmVisibility.None:EmbodiedBodyPresenter.BodyArmVisibility.Both);
        body.PoseWillReset+=bare.CancelImmediately;
        bare.Configure(WorkArm("L",body.WorkLeftArmRenderer),WorkArm("R",body.WorkRightArmRenderer),eye,session.ClockView,BareLease,()=>body.PoseRevision,Validate);
        Check(gloves.ConfigureSession(session.Eva,session.State,session.ClockView,Lease),"Original glove JSON binds without alternate rig.");
        gloves.ConfigureRouteProgress(()=>player.IsOnRoute?player.RouteDistanceMetres:float.NaN);gloves.ConfigureContacts(eye,Lease,Validate);
        player.Pushed+=OnPush;player.Grabbed+=OnGrab;
        captureTarget=new RenderTexture(1280,720,24,RenderTextureFormat.ARGB32);captureTarget.Create();cameraView.targetTexture=captureTarget;cameraView.aspect=1280f/720;
        RenderPipelineManager.endCameraRendering+=OnEndCamera;
        wallMaterial=MakeMaterial("Fixture graphite",new Color(.25f,.31f,.34f));railMaterial=MakeMaterial("Fixture rail",new Color(.68f,.72f,.70f));blockMaterial=MakeMaterial("Fixture blocker",new Color(.65f,.32f,.08f));
        yield return Positive(true,ContactHand.Left,HandContactKind.Push);
        yield return Positive(true,ContactHand.Right,HandContactKind.Push);
        yield return Refusals(true);
        yield return ReversePartial(ContactHand.Left);
        yield return ReversePartial(ContactHand.Right);
        yield return Interrupt(true,"pause");yield return Interrupt(true,"removed");yield return Interrupt(true,"occluded");yield return Interrupt(true,"body-pose");
        yield return UnsafeNeutral();
        // Keep the proven slow-frame regression in this source combination, with no extra fake action.
        yield return ApproachHitch(true,false);
        yield return ApproachHitch(true,true);
        Check(renderedSweepFrames>=20,"At least20 actual rendered frames independently measured across full approaches/returns/cancellations.");
        Check(captures>=3&&lookDownCaptures==2,"Native actual contact and both reverse-path captures recorded.");
        File.WriteAllText(Path.Combine(BarePalmPlaneQa.Output,"whole-skin-summary.json"),
            "{\"renderedFrames\":"+renderedSweepFrames+",\"minimumClearanceMetres\":"+worstClearance.ToString("R",CultureInfo.InvariantCulture)
            +",\"maximumWaypointCorrectionMetres\":"+worstCorrection.ToString("R",CultureInfo.InvariantCulture)+"}");

    }
    BareArmContactRig WorkArm(string side,SkinnedMeshRenderer skin)=>new BareArmContactRig(body.GetWorkBone("upperarm01."+side),body.GetWorkBone("lowerarm01."+side),body.GetWorkBone("wrist."+side),

    IEnumerator Setup(bool useBare,HandContactKind kind)
    {
        watchingSkin=false; bare.CancelImmediately();gloves.ResetToNeutral();body.SetSeatedPose(false);player.ClearMotion();
        if(rail)player.UnregisterHandhold(rail);if(surfaces)Object.DestroyImmediate(surfaces);
        surfaces=new GameObject("Contact QA surfaces");bareMode=useBare;activeKind=kind;
        float distance=useBare?.30f:.60f;
        wall=Box("Actual push wall",new Vector3(0,1.4f,distance+.05f),new Vector3(2.4f,2.8f,.1f),wallMaterial);
        Box("Fixture floor",new Vector3(0,-.06f,0),new Vector3(3,.1f,3),wallMaterial);
        var tube=GameObject.CreatePrimitive(PrimitiveType.Cylinder);tube.name="Registered 15 mm rail";tube.transform.SetParent(surfaces.transform,false);
        tube.transform.SetPositionAndRotation(new Vector3(0,1.65f,distance+.015f),Quaternion.Euler(0,0,90));tube.transform.localScale=new Vector3(.03f,.55f,.03f);tube.GetComponent<Renderer>().sharedMaterial=railMaterial;
        rail=tube.GetComponent<Collider>();rail.gameObject.SetActive(kind==HandContactKind.Grip);wall.gameObject.SetActive(kind==HandContactKind.Push);player.RegisterHandhold(rail);
        blocker=Box("Occluding solid",Vector3.zero,new Vector3(.13f,.13f,.04f),blockMaterial);blocker.gameObject.SetActive(false);
        activeSurface=kind==HandContactKind.Grip?rail:wall;
        player.Teleport(Vector3.zero,0);Physics.SyncTransforms();Done(player.EnterFloat());body.RefreshPresentation();
        yield return Frames(3);yield return StableFrames();
        callbacks=pushes=grabs=0;palmTriangle=null;pendingCapture=null;captureName=null;
        Check(player.IsFloating&&player.ControlAllowed&&session.IsPlayerControlAllowed,"Real EnterFloat at the visible fixture surface with actual control allowed.");
        Check(useBare?body.IsVisible&&!body.IsEvaVisible&&body.VisibleArms==EmbodiedBodyPresenter.BodyArmVisibility.Both&&!gloves.IsVisible
            :body.IsEvaVisible&&body.VisibleArms==EmbodiedBodyPresenter.BodyArmVisibility.None&&gloves.IsVisible,"Exclusive visible arm owner matches authoritative suit state.");
    }
    IEnumerator Positive(bool useBare,ContactHand hand,HandContactKind kind)
    {
        stage=(useBare?"bare":"glove")+"-"+hand+"-"+kind;yield return Setup(useBare,kind);activeHand=hand;
        var target=TargetAt(hand,kind);captureName=stage;
        yield return BeginReach(target,useBare);
        Check(callbacks==0&&pushes==0&&grabs==0&&player.FloatVelocity==Vector3.zero,"Request acceptance is not an immediate gameplay action.");
        yield return WaitFor(()=>callbacks==1,2,"Visible reach/action callback");
        yield return WaitFor(()=>pendingCapture==null,3,"Actual contact-frame camera capture");
        Check(callbacks==1&&(kind==HandContactKind.Push?pushes==1&&grabs==0:grabs==1&&pushes==0),"Exactly one real controller command/event.");
        if(kind==HandContactKind.Grip)
        {
            yield return Seconds(.12f);Check(player.HeldHandhold==rail&&gloves.ContactStage==HandContactStage.Holding,"Actual static registered grip stays held and presented. held="+(player.HeldHandhold?player.HeldHandhold.name:"none")+" stage="+gloves.ContactStage+" validation="+lastValidation+" feet="+player.transform.position.ToString("F6")+" eye="+eye.position.ToString("F6")+" anchor="+player.HeldGripPoint);
            Done(player.Release());yield return WaitFor(()=>gloves.ContactStage==HandContactStage.None,2,"Release retracts hand");
            Check(player.HeldGripPoint==null&&callbacks==1,"Real Release clears the anchor without another callback.");
        }
        else
        {
            yield return Seconds(.12f);Check(Vector3.Dot(player.transform.position-requestedPosition,Vector3.back)>.001f,"Actual successful push produces collision-aware body travel.");
            player.ClearMotion();yield return WaitFor(()=>Stage==HandContactStage.None,2,"Push returns to fitting pose");
            Check(callbacks==1,"Return animation never repeats the action.");
        }
    }
    IEnumerator Refusals(bool useBare)
    {
        stage=(useBare?"bare":"glove")+" refusals";yield return Setup(useBare,HandContactKind.Push);activeHand=ContactHand.Right;
        wall.transform.position+=Vector3.forward*1.2f;Physics.SyncTransforms();var far=TargetAt(activeHand,HandContactKind.Push);
        Check(!StartRequest(far)&&callbacks==0&&Stage==HandContactStage.None,"A genuine distant ray hit refuses before any animation/command.");
        yield return Setup(useBare,HandContactKind.Push);activeHand=ContactHand.Right;var target=TargetAt(activeHand,HandContactKind.Push);
        PlaceBlocker(target);Check(!StartRequest(target)&&callbacks==0&&Stage==HandContactStage.None,"A real intervening solid refuses the formerly visible target.");
    }
    IEnumerator LookDownCaptures()
    {
        stage="bare floating camera fit";yield return Setup(true,HandContactKind.Push);
        Vector3 before=body.VisualRoot.position;Quaternion rotation=body.VisualRoot.rotation;
        foreach(float angle in new[]{60f,80f})
        {
            quietInput.RequestedLook=new Vector2(0,-angle-player.PitchDegrees);
            yield return WaitFor(()=>Math.Abs(player.PitchDegrees+angle)<.01f,2,"Actual Player look-down input");
            yield return Frames(2);
            Check(body.CurrentPose==EmbodiedBodyPresenter.BodyPose.Floating&&Vector3.Distance(before,body.VisualRoot.position)<.0001f
                &&Quaternion.Angle(rotation,body.VisualRoot.rotation)<.001f,"Actual camera looks down while the floating body world pose stays fixed.");
            pendingCapture=new ContactEvidence{name="body-float-lookdown-"+angle.ToString("0",CultureInfo.InvariantCulture),kind="visual-only; no contact action",
                callbackFrame=Time.frameCount,viewPitchDegrees=player.PitchDegrees,bodyVisualOffset=body.GetVisualFeetOffset(body.CurrentPose),bare=true,actualFloating=player.IsFloating,
                rendererVisible=body.IsVisible,realSuitOn=session.Eva.IsSuitOn,playerBefore=player.transform.position,playerAfter=player.transform.position};
            yield return WaitFor(()=>pendingCapture==null,3,"Actual look-down camera capture");
        }
        Check(callbacks==0&&pushes==0&&grabs==0,"Look-down probes issue no physical contact command.");
    }
    IEnumerator Interrupt(bool useBare,string reason)
    {
        stage=(useBare?"bare":"glove")+" interruption "+reason;yield return Setup(useBare,HandContactKind.Push);activeHand=ContactHand.Right;
        var target=TargetAt(activeHand,HandContactKind.Push);yield return BeginReach(target,false);

    bool Nearest(Vector3 from,Vector3 to,out RaycastHit nearest)
    {
        nearest=default;Vector3 delta=to-from;if(delta.magnitude<.0001f)return false;
        float best=float.PositiveInfinity;
        foreach(var hit in Physics.RaycastAll(from,delta.normalized,delta.magnitude+.004f,Physics.DefaultRaycastLayers,QueryTriggerInteraction.Ignore))
            if(!Owned(hit.collider)&&hit.distance<best){nearest=hit;best=hit.distance;}
        return best<float.PositiveInfinity;
    }
    bool Validate(HandContactSample s)
    {

            hand=selected.ToString(),callbackFrame=Time.frameCount,callbackCount=callbacks,grabbedEvents=grabs,pushedEvents=pushes,
            actualFloating=player.IsFloating,realSuitOn=session.Eva.IsSuitOn,rendererVisible=gloves.IsVisible,
            point=player.HeldGripPoint.Value,playerBefore=player.transform.position,playerAfter=player.transform.position,reportedResidual=gloves.ContactErrorMetres};
        yield return WaitFor(()=>pendingCapture==null,3,"Actual post-hitch held capture");
        yield return Seconds(.12f);
        Check(player.HeldHandhold==rail&&Stage==HandContactStage.Holding&&callbacks==1,"Readback following the injected hitch does not retract the static held glove.");
        if(occlude)
        {
            yield return InjectHitch(stage+"-occluded",HandContactStage.Holding,false,()=>PlaceBlocker(requested));
            yield return WaitFor(()=>player.HeldHandhold==null&&Stage==HandContactStage.None,2,"Actual occlusion releases Player and retracts glove");
            Check(callbacks==1,"Occluded slow frame cannot manufacture another grip.");
        }
        else
        {
            Done(player.Release());yield return WaitFor(()=>Stage==HandContactStage.None,2,"Explicit real release still retracts");
            Check(callbacks==1&&player.HeldGripPoint==null,"Release after hitch clears the real anchor once.");
        }
    }
    IEnumerator InjectHitch(string label,HandContactStage wanted,bool remainsValid,Action mutate)
    {
        bool observed=false;
        beginAfterPresenters=()=>
        {
            Check(Stage==wanted,"Exact pre-stall stage is "+wanted+".");
            if(wanted==HandContactStage.Reaching)Check(ReadContactElapsed()<.16f,"At least40ms of visible approach remains before the hitch.");
            Transform[] joints=(bareMode?body.VisualRoot:gloves.ModelRoot).GetComponentsInChildren<Transform>(true);
            Quaternion[] rotations=joints.Select(t=>t.localRotation).ToArray();
            Vector3[] positions=joints.Select(t=>t.localPosition).ToArray();
            var sample=new HitchEvidence{label=label,injectedFrame=Time.frameCount,beforeStage=Stage.ToString(),elapsedBefore=ReadContactElapsed(),
                callbacksBefore=callbacks,validationBefore=lastValidation,heldBefore=player.HeldHandhold?player.HeldHandhold.name:"none",playerBefore=player.transform.position};
            var watch=System.Diagnostics.Stopwatch.StartNew();
            System.Threading.Thread.Sleep(160); // Deliberate Editor-only real frame stall, not mocked delta or production timing.
            watch.Stop();sample.blockedSeconds=watch.Elapsed.TotalSeconds;
            mutate?.Invoke();
            observeHitchAt=Time.frameCount+1;
            afterHitchFrame=()=>
            {
                sample.observedFrame=Time.frameCount;sample.actualDelta=Time.unscaledDeltaTime;sample.afterStage=Stage.ToString();
                sample.elapsedAfter=ReadContactElapsed();sample.callbacksAfter=callbacks;sample.validationAfter=lastValidation;
                sample.heldAfter=player.HeldHandhold?player.HeldHandhold.name:"none";sample.playerAfter=player.transform.position;
                sample.worstJointAngle=joints.Select((t,i)=>Quaternion.Angle(t.localRotation,rotations[i])).Max();
                sample.worstJointPositionMetres=joints.Select((t,i)=>Vector3.Distance(t.localPosition,positions[i])).Max();
                File.WriteAllText(Path.Combine(BarePalmPlaneQa.Output,label+"-hitch-frame.json"),JsonUtility.ToJson(sample,true));
                Check(sample.observedFrame==sample.injectedFrame+1&&sample.actualDelta>.1f&&sample.blockedSeconds>=.15,"Observed the actual immediate >100ms frame after a measured >=150ms stall.");
                Check(callbacks==sample.callbacksBefore,"The actual hitch frame invokes no new contact action.");
                if(remainsValid)
                {
                    Check(Stage==wanted,"A slow frame alone preserves "+wanted+"; current="+Stage+", validation="+lastValidation);
                    Check(Math.Abs(sample.elapsedAfter-sample.elapsedBefore)<.00001f&&sample.worstJointAngle<.01f
                        &&sample.worstJointPositionMetres<.00001f,"Hitch consumes zero animation time and preserves the real joint pose.");
                    Check(Vector3.Distance(sample.playerBefore,sample.playerAfter)<.0001f,"A static slow-frame contact does not create Player movement.");
                }
                else Check(Stage!=wanted,"Genuine changed geometry still cancels on the slow frame; validation="+lastValidation);
                observed=true;
            };
        };
        yield return WaitFor(()=>observed,3,"Injected slow frame observed after real presenters");
    }
    float ReadContactElapsed()
    {
        object presenter=bareMode?(object)bare:gloves;
        string field=bareMode?"elapsed":"contactElapsed";
        return (float)presenter.GetType().GetField(field,BindingFlags.Instance|BindingFlags.NonPublic).GetValue(presenter);
    }
    IEnumerator StableFrames()
    {int stable=0;float until=Time.realtimeSinceStartup+5;while(stable<4){if(Time.realtimeSinceStartup>until)throw new InvalidOperationException("Could not obtain four <80 ms native frames before anticipation.");yield return null;stable=Time.unscaledDeltaTime<=.08f?stable+1:0;}}
    static IEnumerator Frames(int n){for(int i=0;i<n;i++)yield return null;}
    static IEnumerator Seconds(float n){float end=Time.realtimeSinceStartup+n;while(Time.realtimeSinceStartup<end)yield return null;}
    static IEnumerator WaitFor(Func<bool> condition,float seconds,string label)
    {float end=Time.realtimeSinceStartup+seconds;while(!condition()){if(Time.realtimeSinceStartup>=end)throw new TimeoutException(label);yield return null;}}

    }
